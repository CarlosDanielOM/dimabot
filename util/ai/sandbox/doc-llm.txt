# ==========================================
# COMMANDS API
# ==========================================

GET https://api.domdimabot.com/commands/:channelID
authorization: {{AUTH_TOKEN}}
{}
Fetch all commands for a specific channel.

###

POST https://api.domdimabot.com/commands/:channelID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "name": "string",       // Command name for user context
  "cmd": "string",        // Trigger keyword (e.g., "hello") - unique per channel
  "func": "string",       // Function name same as cmd
  "message": "string",    // The response text (supports variables/logic, see below)
  "userLevel": number,    // Permission level (1=Everyone, 2=Sub, 5=VIP, 6=Founder, 7=Mod, 8=Editor, 9=Admin, 10=Broadcaster)
  "cooldown": number,     // Cooldown in seconds (default 10) (min - 5; max - 60)
  "enabled": boolean,     // true/false
  "channel": "string"     // Channel name
}
Create a new chat command.

###

PUT https://api.domdimabot.com/commands/:channelID/:commandID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "message": "string",
  "cooldown": number,
  "userLevel": number,
  "enabled": boolean
}
Update an existing command.

###

DELETE https://api.domdimabot.com/commands/:channelID/:commandID
authorization: {{AUTH_TOKEN}}
{}
Delete a command.

# ==========================================
# COMMAND VARIABLES & LOGIC (for 'message' field)
# ==========================================
# These tags are processed inside the command message.

# Special Syntax:
# $(function args)       - Execute a function
# %(variable)            - Get a temporary variable
# %(variable value)      - Set a temporary variable
# *(condition ? T : F)   - Logic/Ternary (if/else)

# Available Functions ($):
# --- User & Target ---
# $(user)                - Username of the caller
# $(touser)              - Username of target (argument) or caller if no arg
# $(random)              - Random number 0-100
# $(random <max>)        - Random number 0-max
# $(randomuser)          - Random active chatter name
#
# --- Twitch Data ---
# $(twitch.subs)         - Total subscriber count
# $(twitch.viewers)      - Current viewer count
# $(twitch.follows)      - Total follower count
# $(twitch.title)        - Current stream title
# $(twitch.game)         - Current game name
# $(twitch.channel)      - Channel name
#
# --- Stream Management (Requires Mod/Broadcaster) ---
# $(set.game <name>)     - Change stream category
# $(set.title <text>)    - Change stream title
# $(start.poll <args>)   - Start a poll
# $(start.prediction <args>) - Start a prediction
# $(raid <user>)         - Raid a target channel
# $(unraid)              - Cancel an active raid
# $(vip <user>)          - Add VIP role to a user
# $(vip <user> <days>)   - Add VIP role to a user for the amount of days  (max 30)
# $(ban <user> [time])   - Ban a user (time optional in seconds)
#
# --- Counters ---
# $(count)               - Get current count and increment
# $(count <num>)         - Get count and increment by num
# $(scount)              - Simple count increment (+1)
#
# --- AI ---
# $(ai <prompt>)         - Generate a response using AI

# Logic Operators (*):
# ==, !=, >, <, >=, <=, ~= (contains, case-insensitive)
# Example: *($(user) == "admin" ? "Welcome Boss" : "Welcome $(user)")

# ==========================================
# TRIGGERS API (Sound/Video Alerts)
# ==========================================
# Triggers are linked to Channel Point Rewards.
# Note: The 'file' must already exist in the system (uploaded separately).

GET https://api.domdimabot.com/triggers/:channelID
authorization: {{AUTH_TOKEN}}
{}
List all triggers.

###

POST https://api.domdimabot.com/triggers/:channelID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "name": "string",       // Trigger name (becomes Reward title)
  "file": "string",       // Filename of existing media (e.g., "alert.mp3")
  "type": "string",       // "audio" or "video"
  "mediaType": "string",  // MIME type (e.g., "audio/mp3", "video/mp4")
  "cost": number,         // Channel points cost
  "cooldown": number,     // Cooldown in seconds
  "volume": number,       // 0.0 to 1.0 (optional)
  "prompt": "string"      // User input prompt (optional)
}
Create a trigger. This creates a Custom Reward on Twitch automatically.

###

PATCH https://api.domdimabot.com/triggers/:channelID/:triggerID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "name": "string",
  "cost": number,
  "cooldown": number,
  "volume": number,
  "prompt": "string"
}
Update a trigger.

###

DELETE https://api.domdimabot.com/triggers/:channelID/:triggerID
authorization: {{AUTH_TOKEN}}
{}
Delete a trigger (also deletes the associated Twitch Reward).

# ==========================================
# EVENTSUB API (Twitch Event Subscriptions)
# ==========================================

GET https://api.domdimabot.com/eventsubs/:channelID
authorization: {{AUTH_TOKEN}}
{}
List active EventSub subscriptions.

###

POST https://api.domdimabot.com/eventsubs/:channelID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "type": "string",       // e.g., "channel.follow", "stream.online"
  "version": "string",    // e.g., "1", "2"
  "condition": {          // Object matching Twitch requirements for the type
    "broadcaster_user_id": "string"
  }
}
Subscribe to a Twitch EventSub topic.

###

DELETE https://api.domdimabot.com/eventsubs/:channelID/:id
authorization: {{AUTH_TOKEN}}
{}
Unsubscribe from an EventSub topic.

# ==========================================
# REWARDS API (Custom Channel Points)
# ==========================================

GET https://api.domdimabot.com/rewards/:channelID
authorization: {{AUTH_TOKEN}}
{}
List custom rewards managed by the bot.

###

POST https://api.domdimabot.com/rewards/:channelID
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "title": "string",      // Reward title (max 45 chars)
  "cost": number,         // Point cost
  "message": string,      // Message that will be posted in chat when redeem, has same special functions as commands, like: $(user), $(touser), etc.
  "prompt": "string",     // User input prompt (optional)
  "userInput": boolean,   // Require user input? (default false)
  "cooldown": number,     // Global cooldown seconds (0 = none)
  "isEnabled": boolean,   // Default true
  "skipQueue": boolean,   // Skip request queue? (default false)
  "priceIncrease": number // Optional dynamic pricing amount
}
Create a Custom Channel Point Reward on Twitch.

###

PATCH https://api.domdimabot.com/rewards/:channelID/:id
content-type: application/json
authorization: {{AUTH_TOKEN}}
{
  "title": "string",
  "cost": number,
  "message": string,    //Optional, it is not required if not changing the message of the redemption.
  "isEnabled": boolean,
  "cooldown": number
}
Update a reward.

###

DELETE https://api.domdimabot.com/rewards/:channelID/:id
authorization: {{AUTH_TOKEN}}
{}
Delete a custom reward from Twitch and the bot.
